🔌 HOW TO USE GraphQL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Include in PHP:
  <script src="./assets/js/graphql-client.js"></script>

Get Products:
  const data = await GraphQLClient.getProducts();
  console.log(data);

Add to Cart:
  await GraphQLClient.addToCart(productId, quantity);

Check Auth:
  const auth = await GraphQLClient.checkAuthStatus();

🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Problem: "vendor/autoload.php not found"
  → Run: composer install

Problem: "GraphQL API not responding"
  → Check: http://localhost/nivis_lab/status.php
  → Check Browser Console (F12) for errors

Problem: "Port already in use"
  → Change port in XAMPP: httpd.conf
  → Set: Listen 8080
  → Access: http://localhost:8080/nivis_lab/

Problem: "Products not loading"
  → Open Console (F12)
  → Check Network tab for API response
  → Verify graphql-api.php is running
